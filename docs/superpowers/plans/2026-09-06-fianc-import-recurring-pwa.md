# fianc — CSV Import, Recurring Rules, and PWA Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Finish v1 — import a bank CSV with dedupe and one-click undo, post recurring transactions without a scheduler, let a user manage their settings and password, and make the app installable and usable offline for reading.

**Architecture:** The same shape as the core plan. New logic goes into pure modules (`lib/csv.ts`, `lib/recurring.ts`) that are unit-tested without a database; the UI and the server actions stay thin. Recurring transactions are posted lazily by a catch-up call from the app layout, made idempotent by a unique index rather than by a lock. The service worker is generated by Serwist and caches the shell; mutations still require the network, and the UI says so.

**Tech Stack:** Adds `papaparse`, `@serwist/next` + `serwist` to the core stack.

**Spec:** `docs/superpowers/specs/2026-09-06-money-app-design.md`

**Prerequisite:** `docs/superpowers/plans/2026-09-06-fianc-core.md` complete and green. This plan assumes every interface it produced.

## Global Constraints

All constraints from the core plan still apply. Repeated here because they are easy to drop late in a build:

- App name is **fianc** (lowercase). **Lucide icons only, never emoji.**
- Light theme only. `DESIGN.md` tokens for every value: 18px radius on interactive elements, 24px on cards, `#e7000b` for destructive and error states only.
- Money is integer minor units. `amount_minor` is always positive; the sign is `direction`.
- Every query and mutation scoped by `user_id`; another user's row reads as **not found**.
- Server actions return `ActionResult` and never throw across the client boundary.
- Mobile-first, touch targets ≥ 44px.
- Commit after every task.

**One deliberate substitution from the spec:** the spec asks for "a Lighthouse PWA installability check in CI." Task 7 implements that check as a Playwright test that validates the manifest fields and asserts the service worker takes control, rather than adding the Lighthouse CI toolchain. It tests the same property — installability — without a second test runner. If real Lighthouse scoring is wanted later, it can be added alongside.

```ts
import { createHash } from "node:crypto";
import { parseAmount } from "./money";

export type DateOrder = "ymd" | "dmy" | "mdy";

export interface ColumnMap {
  date: string;
  amount: string;
  payee?: string;
  note?: string;
}

export interface ParsedRow {
  date: string;
  amountMinor: number;
  type: "income" | "expense";
  payee: string;
  note: string;
  dedupeHash: string;
}

export interface RowError {
  /** 1-based, matching what the preview table shows the user. */
  row: number;
  message: string;
}

const SEP = /[/\-.]/;

/**
 * Bank exports do not say whether 04/03 is April or March, so the order is
 * chosen by the user in the mapping step. Guessing here would silently
 * mis-date a whole import.
 */
export function normaliseDate(input: string, order: DateOrder): string | null {
  const trimmed = input.trim();
  if (trimmed === "") return null;

  const parts = trimmed.split(SEP).map((p) => p.trim());
  if (parts.length !== 3 || parts.some((p) => !/^\d+$/.test(p))) return null;

  let y: string, m: string, d: string;
  if (order === "ymd") [y, m, d] = parts;
  else if (order === "dmy") [d, m, y] = parts;
  else [m, d, y] = parts;

  if (y.length !== 4) return null;

  const year = Number(y);
  const month = Number(m);
  const day = Number(d);
  if (month < 1 || month > 12 || day < 1 || day > 31) return null;

  // Round-trip through UTC so 31 February is rejected rather than rolled over.
  const date = new Date(Date.UTC(year, month - 1, day));
  if (
    date.getUTCFullYear() !== year ||
    date.getUTCMonth() !== month - 1 ||
    date.getUTCDate() !== day
  ) {
    return null;
  }

  return `${y}-${m.padStart(2, "0")}-${d.padStart(2, "0")}`;
}

export function dedupeHash(
  accountId: string,
  date: string,
  amountMinor: number,
  payee: string,
): string {
  return createHash("sha256")
    .update(`${accountId}|${date}|${amountMinor}|${payee.trim().toLowerCase()}`)
    .digest("hex");
}

export function mapRows(
  rows: Record<string, string>[],
  map: ColumnMap,
  opts: { order: DateOrder; accountId: string },
): { rows: ParsedRow[]; errors: RowError[] } {
  const out: ParsedRow[] = [];
  const errors: RowError[] = [];
  const seen = new Set<string>();

  rows.forEach((raw, i) => {
    const rowNumber = i + 1;
    const dateCell = (raw[map.date] ?? "").trim();
    const amountCell = (raw[map.amount] ?? "").trim();

    // A trailing blank line is not an error.
    if (dateCell === "" && amountCell === "") return;

    const date = normaliseDate(dateCell, opts.order);
    if (!date) {
      errors.push({ row: rowNumber, message: "Could not read the date" });
      return;
    }

    const negative =
      amountCell.startsWith("-") ||
      (amountCell.startsWith("(") && amountCell.endsWith(")"));
    const bare = amountCell.replace(/^[-(]/, "").replace(/\)$/, "");

    const amount = parseAmount(bare);
    if (!amount.ok) {
      errors.push({ row: rowNumber, message: amount.error });
      return;
    }

    const payee = (map.payee ? (raw[map.payee] ?? "") : "").trim();
    const note = (map.note ? (raw[map.note] ?? "") : "").trim();
    const hash = dedupeHash(opts.accountId, date, amount.value, payee);

    if (seen.has(hash)) {
      errors.push({
        row: rowNumber,
        message: "Duplicate of an earlier row in this file",
      });
      return;
    }
    seen.add(hash);

    out.push({
      date,
      amountMinor: amount.value,
      type: negative ? "expense" : "income",
      payee,
      note,
      dedupeHash: hash,
    });
  });

  return { rows: out, errors };
}
```

```ts
"use server";

import { z } from "zod";
import { and, eq, sql } from "drizzle-orm";
import { revalidatePath } from "next/cache";
import { db } from "@/lib/db";
import { accounts, importBatches, transactions } from "@/lib/db/schema";
import { requireUser } from "@/lib/auth/guard";
import { directionFor } from "@/lib/money";
import { ok, fail, type ActionResult } from "@/lib/action-result";

const rowSchema = z.object({
  date: z.string().regex(/^\d{4}-\d{2}-\d{2}$/),
  amountMinor: z.number().int().positive(),
  type: z.enum(["income", "expense"]),
  payee: z.string(),
  note: z.string(),
  dedupeHash: z.string().regex(/^[0-9a-f]{64}$/),
});

const commitSchema = z.object({
  accountId: z.string().uuid(),
  filename: z.string().min(1).max(255),
  rows: z.array(rowSchema).min(1, "Nothing to import").max(5000),
});

export type CommitImportInput = z.infer<typeof commitSchema>;

/**
 * The whole batch lands in one database transaction, so a failure part-way
 * leaves nothing behind. The unique index on (user_id, dedupe_hash) is the
 * real defence against a double import; the preview is a convenience.
 */
export async function commitImport(
  input: CommitImportInput,
): Promise<ActionResult<{ batchId: string; inserted: number; skipped: number }>> {
  const user = await requireUser();
  const parsed = commitSchema.safeParse(input);
  if (!parsed.success) return fail("That import is not valid");

  const { accountId, filename, rows } = parsed.data;

  const [account] = await db
    .select({ id: accounts.id })
    .from(accounts)
    .where(and(eq(accounts.id, accountId), eq(accounts.userId, user.id)))
    .limit(1);
  if (!account) return fail("Account not found");

  const result = await db.transaction(async (tx) => {
    const [batch] = await tx
      .insert(importBatches)
      .values({
        userId: user.id,
        accountId,
        filename,
        rowCount: rows.length,
      })
      .returning({ id: importBatches.id });

    const inserted = await tx
      .insert(transactions)
      .values(
        rows.map((r) => ({
          userId: user.id,
          accountId,
          categoryId: null,
          type: r.type,
          direction: directionFor(r.type),
          amountMinor: r.amountMinor,
          date: r.date,
          payee: r.payee,
          note: r.note,
          importBatchId: batch.id,
          dedupeHash: r.dedupeHash,
        })),
      )
      // The dedupe index is PARTIAL (`where dedupe_hash is not null`), so
      // Postgres needs the same predicate to infer it. Without targetWhere
      // this raises "no unique or exclusion constraint matches".
      .onConflictDoNothing({
        target: [transactions.userId, transactions.dedupeHash],
        targetWhere: sql`${transactions.dedupeHash} is not null`,
      })
      .returning({ id: transactions.id });

    return {
      batchId: batch.id,
      inserted: inserted.length,
      skipped: rows.length - inserted.length,
    };
  });

  revalidatePath("/transactions");
  revalidatePath("/dashboard");
  revalidatePath("/import");
  return ok(result);
}

/** Deleting the batch cascades to its transactions. */
export async function undoImport(batchId: string): Promise<ActionResult> {
  const user = await requireUser();
  const [row] = await db
    .delete(importBatches)
    .where(and(eq(importBatches.id, batchId), eq(importBatches.userId, user.id)))
    .returning({ id: importBatches.id });

  if (!row) return fail("Import not found");

  revalidatePath("/transactions");
  revalidatePath("/dashboard");
  revalidatePath("/import");
  return ok();
}
```

```tsx
"use client";

import { useState, useTransition } from "react";
import Papa from "papaparse";
import { Upload, TriangleAlert } from "lucide-react";
import { toast } from "sonner";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { Card, CardContent } from "@/components/ui/card";
import {
  Select,
  SelectContent,
  SelectItem,
  SelectTrigger,
  SelectValue,
} from "@/components/ui/select";
import { formatAmount } from "@/lib/money";
import { mapRows, type DateOrder, type ParsedRow, type RowError } from "@/lib/csv";
import type { Account } from "@/lib/db/schema";
import { commitImport, undoImport } from "./actions";

type Stage = "upload" | "map" | "done";

export function ImportWizard({
  accounts,
  currency,
}: {
  accounts: Account[];
  currency: string;
}) {
  const [stage, setStage] = useState<Stage>("upload");
  const [filename, setFilename] = useState("");
  const [headers, setHeaders] = useState<string[]>([]);
  const [raw, setRaw] = useState<Record<string, string>[]>([]);
  const [accountId, setAccountId] = useState(accounts[0]?.id ?? "");
  const [order, setOrder] = useState<DateOrder>("ymd");
  const [cols, setCols] = useState({ date: "", amount: "", payee: "", note: "" });
  const [result, setResult] = useState<
    { batchId: string; inserted: number; skipped: number } | null
  >(null);
  const [pending, startTransition] = useTransition();

  function onFile(file: File) {
    Papa.parse<Record<string, string>>(file, {
      header: true,
      skipEmptyLines: true,
      complete: ({ data, meta }) => {
        const fields = meta.fields ?? [];
        setHeaders(fields);
        setRaw(data);
        setFilename(file.name);
        setCols({
          date: fields.find((f) => /date/i.test(f)) ?? fields[0] ?? "",
          amount: fields.find((f) => /amount|value/i.test(f)) ?? fields[1] ?? "",
          payee:
            fields.find((f) => /desc|payee|name|detail/i.test(f)) ?? "",
          note: "",
        });
        setStage("map");
      },
      error: () => toast.error("That file could not be parsed"),
    });
  }

  let parsed: { rows: ParsedRow[]; errors: RowError[] } = { rows: [], errors: [] };
  if (stage === "map" && cols.date && cols.amount && accountId) {
    parsed = mapRows(
      raw,
      {
        date: cols.date,
        amount: cols.amount,
        payee: cols.payee || undefined,
        note: cols.note || undefined,
      },
      { order, accountId },
    );
  }

  function commit() {
    startTransition(async () => {
      const res = await commitImport({
        accountId,
        filename,
        rows: parsed.rows,
      });
      if (!res.ok) {
        toast.error(res.error);
        return;
      }
      setResult(res.data);
      setStage("done");
    });
  }

  if (stage === "upload") {
    return (
      <Card>
        <CardContent className="p-5 space-y-3">
          <Label htmlFor="file">Bank CSV export</Label>
          <Input
            id="file"
            type="file"
            accept=".csv,text/csv"
            className="h-11"
            onChange={(e) => {
              const file = e.target.files?.[0];
              if (file) onFile(file);
            }}
          />
          <p className="text-muted-foreground">
            Nothing is saved until you review the preview.
          </p>
        </CardContent>
      </Card>
    );
  }

  if (stage === "done" && result) {
    return (
      <Card>
        <CardContent className="p-5 space-y-3">
          <p className="text-subheading font-medium">
            Imported {result.inserted} transaction
            {result.inserted === 1 ? "" : "s"}
          </p>
          {result.skipped > 0 && (
            <p className="text-muted-foreground">
              Skipped {result.skipped} already in fianc.
            </p>
          )}
          <Button
            variant="outline"
            disabled={pending}
            onClick={() =>
              startTransition(async () => {
                const res = await undoImport(result.batchId);
                if (res.ok) {
                  toast.success("Import undone");
                  setStage("upload");
                  setResult(null);
                } else {
                  toast.error(res.error);
                }
              })
            }
          >
            Undo this import
          </Button>
        </CardContent>
      </Card>
    );
  }

  return (
    <div className="space-y-5">
      <Card>
        <CardContent className="p-5 grid gap-3 md:grid-cols-2">
          <Field label="Account">
            <Select value={accountId} onValueChange={setAccountId}>
              <SelectTrigger className="h-11"><SelectValue /></SelectTrigger>
              <SelectContent>
                {accounts.map((a) => (
                  <SelectItem key={a.id} value={a.id}>{a.name}</SelectItem>
                ))}
              </SelectContent>
            </Select>
          </Field>

          <Field label="Date order">
            <Select value={order} onValueChange={(v) => setOrder(v as DateOrder)}>
              <SelectTrigger className="h-11"><SelectValue /></SelectTrigger>
              <SelectContent>
                <SelectItem value="ymd">Year, month, day</SelectItem>
                <SelectItem value="dmy">Day, month, year</SelectItem>
                <SelectItem value="mdy">Month, day, year</SelectItem>
              </SelectContent>
            </Select>
          </Field>

          {(["date", "amount", "payee", "note"] as const).map((key) => (
            <Field key={key} label={`${key[0].toUpperCase()}${key.slice(1)} column`}>
              <Select
                value={cols[key] || "__none"}
                onValueChange={(v) =>
                  setCols((c) => ({ ...c, [key]: v === "__none" ? "" : v }))
                }
              >
                <SelectTrigger className="h-11"><SelectValue /></SelectTrigger>
                <SelectContent>
                  <SelectItem value="__none">Not mapped</SelectItem>
                  {headers.map((h) => (
                    <SelectItem key={h} value={h}>{h}</SelectItem>
                  ))}
                </SelectContent>
              </Select>
            </Field>
          ))}
        </CardContent>
      </Card>

      {parsed.errors.length > 0 && (
        <Card>
          <CardContent className="p-5">
            <p className="flex items-center gap-2 text-destructive mb-2">
              <TriangleAlert className="size-4" strokeWidth={1.5} />
              {parsed.errors.length} row
              {parsed.errors.length === 1 ? "" : "s"} will be skipped
```

```ts
export interface Schedule {
  freq: "daily" | "weekly" | "monthly" | "yearly";
  interval: number;
  /** Weekly: 0 (Sunday) to 6. Monthly and yearly: day of month, 1 to 31. */
  anchorDay: number;
  /** Yearly only: 1 to 12. Falls back to the start date's month. */
  anchorMonth?: number | null;
  startDate: string;
  endDate?: string | null;
}

/** Runaway guard: a decade of daily occurrences is far past anything real. */
const MAX_OCCURRENCES = 4000;

function toUtc(iso: string): Date {
  const [y, m, d] = iso.split("-").map(Number);
  return new Date(Date.UTC(y, m - 1, d));
}

function fmt(date: Date): string {
  return date.toISOString().slice(0, 10);
}

function daysInMonth(year: number, monthIndex: number): number {
  return new Date(Date.UTC(year, monthIndex + 1, 0)).getUTCDate();
}

/** Month arithmetic that clamps rather than rolling over: 31 Jan + 1 month = 28 Feb. */
function clampedMonth(year: number, monthIndex: number, day: number): Date {
  const normalisedYear = year + Math.floor(monthIndex / 12);
  const normalisedMonth = ((monthIndex % 12) + 12) % 12;
  const d = Math.min(day, daysInMonth(normalisedYear, normalisedMonth));
  return new Date(Date.UTC(normalisedYear, normalisedMonth, d));
}

/**
 * Every occurrence from `startDate` through `through`, excluding anything on
 * or before `after` (the rule's `last_posted_date`).
 *
 * Pure and deterministic: the caller decides "today", which is what makes the
 * lazy catch-up in Task 4 testable and idempotent.
 */
export function occurrencesThrough(
  schedule: Schedule,
  through: string,
  after?: string | null,
): string[] {
  const { freq, anchorDay, anchorMonth, startDate, endDate } = schedule;
  const interval = Math.max(1, Math.floor(schedule.interval));

  const start = toUtc(startDate);
  const limit = toUtc(through);
  const hardEnd = endDate ? toUtc(endDate) : null;
  const cursor = after ? toUtc(after) : null;

  const out: string[] = [];
  let current: Date;
  let step = 0;

  if (freq === "weekly") {
    // First occurrence is the anchored weekday on or after the start date.
    const shift = (anchorDay - start.getUTCDay() + 7) % 7;
    current = new Date(start.getTime() + shift * 86_400_000);
  } else if (freq === "monthly") {
    current = clampedMonth(
      start.getUTCFullYear(),
      start.getUTCMonth(),
      anchorDay,
    );
    if (current < start) {
      current = clampedMonth(
        start.getUTCFullYear(),
        start.getUTCMonth() + interval,
        anchorDay,
      );
      step = 1;
    }
  } else if (freq === "yearly") {
    const month = (anchorMonth ?? start.getUTCMonth() + 1) - 1;
    current = clampedMonth(start.getUTCFullYear(), month, anchorDay);
    if (current < start) {
      current = clampedMonth(start.getUTCFullYear() + interval, month, anchorDay);
      step = 1;
    }
  } else {
    current = start;
  }

  while (current <= limit && out.length < MAX_OCCURRENCES) {
    if (hardEnd && current > hardEnd) break;
    if (!cursor || current > cursor) out.push(fmt(current));

    step += 1;
    if (freq === "daily") {
      current = new Date(start.getTime() + step * interval * 86_400_000);
    } else if (freq === "weekly") {
      const first = toUtc(out[0] ?? fmt(current));
      current = new Date(first.getTime() + step * interval * 7 * 86_400_000);
    } else if (freq === "monthly") {
      current = clampedMonth(
        start.getUTCFullYear(),
        start.getUTCMonth() + step * interval,
        anchorDay,
      );
    } else {
      const month = (anchorMonth ?? start.getUTCMonth() + 1) - 1;
      current = clampedMonth(
        start.getUTCFullYear() + step * interval,
        month,
        anchorDay,
      );
    }
  }

  return out;
}
```

```ts
import "server-only";
import { and, asc, eq } from "drizzle-orm";
import { db } from "@/lib/db";
import { recurringRules, transactions, type RecurringRule } from "@/lib/db/schema";
import { occurrencesThrough } from "@/lib/recurring";
import { directionFor } from "@/lib/money";

export async function listRules(userId: string): Promise<RecurringRule[]> {
  return db
    .select()
    .from(recurringRules)
    .where(eq(recurringRules.userId, userId))
    .orderBy(asc(recurringRules.createdAt));

/**
 * Posts every occurrence due on or before `today` for every active rule.
 *
 * There is no scheduler: this runs when the app is opened. Two concurrent
 * loads cannot double-post because the unique index on
 * (recurring_rule_id, occurrence_date) rejects the second insert, and
 * `onConflictDoNothing` turns that rejection into a no-op rather than an error.
 */
export async function postDueOccurrences(
  userId: string,
  today: string = new Date().toISOString().slice(0, 10),
): Promise<number> {
  const rules = await db
    .select()
    .from(recurringRules)
    .where(
      and(eq(recurringRules.userId, userId), eq(recurringRules.paused, false)),
    );

  let posted = 0;

  for (const rule of rules) {
    const dates = occurrencesThrough(
      {
        freq: rule.freq,
        interval: rule.interval,
        anchorDay: rule.anchorDay,
        anchorMonth: rule.anchorMonth,
        startDate: rule.startDate,
        endDate: rule.endDate,
      },
      today,
      rule.lastPostedDate,
    );
    if (dates.length === 0) continue;

    const inserted = await db
      .insert(transactions)
      .values(
        dates.map((date) => ({
          userId,
          accountId: rule.accountId,
          categoryId: rule.categoryId,
          type: rule.type,
          direction: directionFor(rule.type),
          amountMinor: rule.amountMinor,
          date,
          payee: rule.payee,
          note: rule.note,
          recurringRuleId: rule.id,
          occurrenceDate: date,
        })),
      )
      .onConflictDoNothing({
        target: [transactions.recurringRuleId, transactions.occurrenceDate],
      })
      .returning({ id: transactions.id });

    posted += inserted.length;

    await db
      .update(recurringRules)
      .set({ lastPostedDate: dates[dates.length - 1] })
      .where(eq(recurringRules.id, rule.id));
  }

  return posted;
}
```

```ts
"use server";

import { z } from "zod";
import { eq, and, ne } from "drizzle-orm";
import { revalidatePath } from "next/cache";
import { redirect } from "next/navigation";
import { db } from "@/lib/db";
import { users, sessions } from "@/lib/db/schema";
import { requireUser } from "@/lib/auth/guard";
import { hashPassword, verifyPassword } from "@/lib/auth/password";
import { ok, fail, type ActionResult } from "@/lib/action-result";

const profileInput = z.object({
  name: z.string().trim().min(1, "Enter a name"),
  baseCurrency: z
    .string()
    .trim()
    .toUpperCase()
    .regex(/^[A-Z]{3}$/, "Use a three-letter currency code, e.g. USD"),
});

export async function updateProfile(
  _prev: ActionResult | null,
  formData: FormData,
): Promise<ActionResult> {
  const user = await requireUser();
  const parsed = profileInput.safeParse(Object.fromEntries(formData));
  if (!parsed.success) {
    return fail("Check the form", parsed.error.flatten().fieldErrors);
  }

  // Reject a code Intl cannot format rather than breaking every amount later.
  try {
    new Intl.NumberFormat("en-US", {
      style: "currency",
      currency: parsed.data.baseCurrency,
    }).format(1);
  } catch {
    return fail("Check the form", {
      baseCurrency: ["That currency code is not recognised"],
    });
  }

  await db.update(users).set(parsed.data).where(eq(users.id, user.id));
  revalidatePath("/settings");
  revalidatePath("/dashboard");
  return ok();
}

const passwordInput = z.object({
  current: z.string().min(1, "Enter your current password"),
  next: z.string().min(10, "Use at least 10 characters"),
});

/**
 * Changing the password invalidates every other session, then signs this one
 * out too — the simplest correct behaviour, and cheap when there are two users.
 */
export async function changePassword(
  _prev: ActionResult | null,
  formData: FormData,
): Promise<ActionResult> {
  const user = await requireUser();
  const parsed = passwordInput.safeParse(Object.fromEntries(formData));
  if (!parsed.success) {
    return fail("Check the form", parsed.error.flatten().fieldErrors);
  }

  if (!(await verifyPassword(user.passwordHash, parsed.data.current))) {
    return fail("Check the form", { current: ["That password is not correct"] });
  }

  await db
    .update(users)
    .set({ passwordHash: await hashPassword(parsed.data.next) })
    .where(eq(users.id, user.id));

  await db.delete(sessions).where(eq(sessions.userId, user.id));
  redirect("/login");
}
```

```ts
"use client";

import { useSyncExternalStore } from "react";

function subscribe(onChange: () => void) {
  window.addEventListener("online", onChange);
  window.addEventListener("offline", onChange);
  return () => {
    window.removeEventListener("online", onChange);
    window.removeEventListener("offline", onChange);
  };
}

export function useOnline(): boolean {
  return useSyncExternalStore(
    subscribe,
    () => window.navigator.onLine,
    () => true, // server snapshot: assume online, the common case
  );
}
```

```tsx
"use client";

import { useEffect, useState } from "react";
import { Download, X } from "lucide-react";
import { Button } from "@/components/ui/button";

interface BeforeInstallPromptEvent extends Event {
  prompt: () => Promise<void>;
  userChoice: Promise<{ outcome: "accepted" | "dismissed" }>;
}

const DISMISSED_KEY = "fianc:install-dismissed";

export function InstallPrompt() {
  const [event, setEvent] = useState<BeforeInstallPromptEvent | null>(null);

  useEffect(() => {
    // Safari never fires this; iOS install is documented in Settings instead.
    if (localStorage.getItem(DISMISSED_KEY)) return;
    const onPrompt = (e: Event) => {
      e.preventDefault();
      setEvent(e as BeforeInstallPromptEvent);
    };
    window.addEventListener("beforeinstallprompt", onPrompt);
    return () => window.removeEventListener("beforeinstallprompt", onPrompt);
  }, []);

  if (!event) return null;

  function dismiss() {
    localStorage.setItem(DISMISSED_KEY, "1");
    setEvent(null);
  }

  return (
    <div className="mb-4 flex items-center gap-3 rounded-card border border-border bg-card p-4 shadow-card">
      <Download className="size-4 shrink-0" strokeWidth={1.5} />
      <p className="flex-1 min-w-0">Install fianc for one-tap access.</p>
      <Button
        size="sm"
        onClick={async () => {
          await event.prompt();
          await event.userChoice;
          dismiss();
        }}
      >
        Install
      </Button>
      <Button size="icon" variant="ghost" onClick={dismiss} aria-label="Dismiss">
        <X className="size-4" strokeWidth={1.5} />
      </Button>
    </div>
  );
}
```
